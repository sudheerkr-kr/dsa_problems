class Solution {
  public:
  void dfs(int row, int col  , vector<vector<int>>&visited, vector<vector<char>>&grid , int delrow[],int delcol[]){
      visited[row][col]=1;
      int n=grid.size();
      int m=grid[0].size();
      //check for top, right, bottom ,left
      for(int i=0 ;i<4; i++){
          int nrow=row+delrow[i];
          int ncol= col+delcol[i];
          if(nrow>=0 &&nrow<n && ncol>=0 && ncol<m && !visited[nrow][ncol] &&grid[nrow][ncol]=='O'){
              dfs(nrow,ncol,visited,grid,delrow,delcol);
          }
      }
      
  }
    vector<vector<char>> fill(vector<vector<char>>& grid) {
        // Code here
        int n= grid.size();
        int m= grid[0].size();
        vector<vector<int>>visited(n,vector(m,0));
      
        int delrow[] ={-1,0,1,0};
        int delcol[] ={0,1,0,-1};
          //traverse frist row and last col
        for(int j=0; j<m; j++){
            //first row
            if(!visited[0][j]&&grid[0][j] =='O'){
                dfs(0,j,visited,grid, delrow, delcol);
            }
            //last row
            if(!visited[n-1][j]&&grid[n-1][j] =='O'){
                dfs(n-1,j,visited,grid, delrow, delcol);
            }
            
        }
        for(int i=0; i<n; i++){
           //first col
            if(!visited[i][0]&&grid[i][0] =='O'){
                dfs(i,0,visited,grid, delrow, delcol);
            }
            //last col
            if(!visited[i][m-1]&&grid[i][m-1] =='O'){
                dfs(i,m-1,visited,grid, delrow, delcol);
            } 
        }
        for(int i=0; i<n; i++){
            for(int j=0; j<m; j++){
                if(!visited[i][j] && grid[i][j]=='O'){
                    grid[i][j]='X';
                }
            }
        }
        
        return grid;
    }
};
